The FCW self-check (`0xB1A0` mod-10 Luhn, `0xB140` mod-11) tells the *emulator* to validate a check digit before allowing Field Exit. Operator error `0x15` is generated terminal-side. If the host trusts the terminal (1980s assumption), strip the FCW → invalid account numbers persist. We don't *delete* (length change breaks things) — we *replace* with `0x8000` (FCW marker, no payload — emulators skip unknown FCW types harmlessly).

```python
"""
IBM i message-ID detection — the abend/oracle harvester.

Direct port of hack3270's ESMFingerprinter (attacks/esm_passive.py)
for the IBM i CPF/MCH/RNQ message-ID space. Observer-attached.

The CPF1107/CPF1120 pair is the username-enumeration oracle:
  CPF1107 "Password not correct" → userid IS valid
  CPF1120 "User does not exist"  → userid is NOT valid
Same disclosure as 3270's DFHCE3530/3532, but always present —
IBM i can't be configured to suppress these like CICS TS 5.1+ can.

MCH1202 is the abend you WANT to see during fuzzing — it means your
numeric injection landed in a packed/zoned field and the program
tried to use it. The Display Program Messages panel that follows
leaks the program/library/statement number.
"""
import re
from hackterm_core import Screen


_RULES: list[tuple[re.Pattern, str, str]] = [
    (re.compile(r'\bMCH1202\b'),  'decimal_data_error',
     'Decimal data error — packed/zoned field got non-numeric bytes'),
    (re.compile(r'\bMCH3601\b'),  'null_pointer',
     'Pointer not set — RPG used unset basing pointer'),
    (re.compile(r'\bCPF1107\b'),  'username_oracle',
     'Password not correct — USERID IS VALID (oracle)'),
    (re.compile(r'\bCPF1120\b'),  'username_oracle',
     'User does not exist (the other half of the oracle)'),
    (re.compile(r'\bCPF1394\b'),  'account_state_leak',
     'User profile cannot sign on — disabled or restricted'),
    (re.compile(r'\bCPF2204\b'),  'object_not_found',
     'User profile not found at object level (auth-layer distinction)'),
    (re.compile(r'\bCPD0078\b'),  'dds_validation_fired',
     'DDS VALUES/RANGE/COMP rejected — host re-validates (good for them)'),
    (re.compile(r'\bRNQ\d{4}\b'), 'rpg_inquiry',
     'RPG inquiry message — program hit unexpected condition'),
    (re.compile(r'\bRNX\d{4}\b'), 'rpg_exception',
     'RPG exception — uncaught error in ILE program'),
    (re.compile(r'Display Program Messages'), 'stack_trace_leak',
     'DSPPGMMSG panel — leaks program/library/statement number'),
]

# DSPPGMMSG panel format — extract program/library names
_PROG_RE = re.compile(r'Program\s*\.+\s*:\s*(\w+)')
_LIB_RE  = re.compile(r'Library\s*\.+\s*:\s*(\w+)')


class IBMiMessageDetector:
    """Passive observer. Watches every s2c screen for IBM i message IDs.

    Populates self.findings: dict[str, list[str]] mapping finding key
    to the list of matched message texts. Same shape as hack3270's
    ESMFingerprinter.findings.
    """

    def __init__(self, protocol):
        self.protocol = protocol
        self.findings: dict[str, list[str]] = {}

    def attach(self, daemon):
        daemon.add_observer(self._observe)

    def _observe(self, data: bytes, direction: str):
        if direction != "s2c":
            return
        screen = self.protocol.parse(data)
        text = screen.text
        for pattern, key, _desc in _RULES:
            for match in pattern.finditer(text):
                self.findings.setdefault(key, []).append(match.group(0))
        # Special: extract program/library from DSPPGMMSG
        if 'stack_trace_leak' in self.findings:
```

```python
# === STRPCCMD canary (CVE-2005-0868) — defensive ===

# WSF (Write Structured Field) with class 0xD9. Within that, the PC
# Organizer types live in a specific range, but ANY structured field
# from a 5250 application is suspicious — normal RPG/COBOL apps don't
# emit them. Match broadly, alert on all.
_PCO_PATTERN = re.compile(rb'\x04\xf3..\xd9')


class STRPCCMDCanary:
    """Defensive: detect host→client RCE attempt via STRPCCMD.

    CVE-2005-0868 — STRPCO+STRPCCMD sends a 5250 structured-field escape
    that makes IBM iAccess/PCOMM execute arbitrary commands on the
    operator's PC. Demonstrated for backdoor creation via REXEC.

    A compromised IBM i (or rogue admin) would use the proxy connection
    against the operator. This is the ONLY host→client RCE in the protocol.
    No other tool watches for it.
    """

    def __init__(self):
        self.alerts: list[dict] = []

    def attach(self, daemon):
        daemon.add_observer(self._observe)

    def _observe(self, data: bytes, direction: str):
        if direction != "s2c":
            return
        match = _PCO_PATTERN.search(data)
        if match:
            # Extract a window around the match for context
            start = max(0, match.start() - 4)
            end = min(len(data), match.end() + 32)
            self.alerts.append({
                "raw_hex": data[start:end].hex(),
                "match_offset": match.start(),
                "warning": (
                    "STRPCCMD escape sequence detected — host attempting "
                    "client-side command execution (CVE-2005-0868). "
                    "If your emulator is IBM iAccess/PCOMM, it may have "
                    "executed the embedded command."
                ),
            })
```

```python
"""
Surface analyzers — pure functions on Screen. No wire traffic.

The data was always in last_screen.fields — parse() reads field content
regardless of the hidden attr. These functions surface it cleanly.

Live finding (Mels Cargo, 2026-04-08): the hidden "99) Delete Order
History" text was visible in screen.text the whole time. The proxy
just didn't have a one-call extract.
"""
from hackterm_core import Screen, EbcdicCodec


def get_hidden_content(screen: Screen, codec: EbcdicCodec) -> list[dict]:
    """Just the NONDISP fields, decoded.

    Returns: [{field_idx, row, col, length, ebcdic_hex, text}, ...]
    """
    out = []
    for i, f in enumerate(screen.fields):
        if not f.hidden:
            continue
        out.append({
            "field_idx": i,
            "row": f.row, "col": f.col, "length": f.length,
            "ebcdic_hex": f.content.hex(),
            "text": codec.to_ascii(f.content),
        })
    return out


def tamper_map(screen: Screen, codec: EbcdicCodec) -> list[dict]:
    """All protected fields ranked by interestingness.

    Heuristics (additive scoring):
      +4  hidden                — suspicious by definition
      +3  contains digits       — price/qty/code
      +3  length 1-4            — flag/PIN/code
      +2  length matches input  — possible echo-back
      -2  all blanks            — boring
      -2  length > 50           — decorative output

    Returns: sorted by score descending.
    [{field_idx, row, col, length, score, reasons, current_value}, ...]
    """
    input_lengths = {f.length for f in screen.fields if not f.protected}

    ranked = []
    for i, f in enumerate(screen.fields):
        if not f.protected:
            continue
        text = codec.to_ascii(f.content)
        score = 0
        reasons: list[str] = []

        if f.hidden:
            score += 4
            reasons.append("HIDDEN")
        if any(c.isdigit() for c in text):
            score += 3
            reasons.append("contains digits")
        if 1 <= f.length <= 4:
            score += 3
            reasons.append("short (flag/code length)")
        if f.length in input_lengths:
            score += 2
            reasons.append(f"length matches an input field ({f.length})")
        if not text.strip():
            score -= 2
            reasons.append("blank")
        if f.length > 50:
            score -= 2
            reasons.append("long (decorative)")

        ranked.append({
            "field_idx": i,
            "row": f.row, "col": f.col, "length": f.length,
            "score": score,
            "reasons": reasons,
            "current_value": text[:40],
        })

    ranked.sort(key=lambda x: -x["score"])
    return ranked
```
